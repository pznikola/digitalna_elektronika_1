\documentclass[tikz,border=5pt]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern,amsmath,amssymb}
\usepackage[american]{circuitikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{circuits.logic.US,positioning,calc,arrows.meta,patterns,shapes.geometric}
\definecolor{Primary}{HTML}{1F77B4}
\begin{document}
\begin{tikzpicture}[circuit logic US,thick]
\node[xor gate US,draw,logic gate inputs=nn](x0)at(1,4){};
\node[and gate US,draw,logic gate inputs=nn](c0)at(1,2.3){};
\node[xor gate US,draw,logic gate inputs=nn](p1)at(1,.6){};
\node[and gate US,draw,logic gate inputs=nn](g1)at(1,-1.1){};
\foreach \n/\a/\b in {x0/a_0/b_0,c0/a_0/b_0,p1/a_1/b_1,g1/a_1/b_1}{\draw (\n.input 1)--++(-.3,0)--++(-.6,.25)node[left]{$\a$};\draw (\n.input 2)--++(-.3,0)--++(-.6,-.25)node[left]{$\b$};}
\draw (x0.output)--(8.1,4) node[right]{$s_0$};
% Align the receiving pins with the source outputs, not the gate centers.
\node[xor gate US,draw,logic gate inputs=nn,anchor=input 1](x1)at(4.65,2.3){};
\node[and gate US,draw,logic gate inputs=nn,anchor=input 2](cp)at(4.65,.6){};
% Carry reaches the upper input of both gates.
\coordinate (carry-branch) at (2.5,2.3);
\coordinate (carry-bend) at (2.5,2.3 |- cp.input 1);
\coordinate (crossing) at (3.5,.6 |- cp.input 1);
\draw (c0.output)--(carry-branch);
\draw (carry-branch)--(x1.input 1);
\draw (carry-branch)--(carry-bend)--([xshift=-.12cm]crossing)
  arc[start angle=180,end angle=0,radius=.12]--(cp.input 1);
\fill (carry-branch) circle (1.3pt);
% The XOR output reaches both lower inputs. A bridge keeps the crossing separate.
\coordinate (propagate-branch) at (3.5,.6);
\draw (p1.output)--(propagate-branch);
\draw (propagate-branch) |- (x1.input 2);
\draw (propagate-branch)--(cp.input 2);
\fill (propagate-branch) circle (1.3pt);
\draw (x1.output)--(8.1,0 |- x1.output) node[right]{$s_1$};
\node[or gate US,draw,logic gate inputs=nn](s2)at(7,-.4){};
\draw (cp.output)--(6.1,0 |- cp.output) |- (s2.input 1);
\draw (g1.output)--(5.7,-1.1) |- (s2.input 2);
\draw (s2.output)--(8.1,-.4) node[right]{$s_2$};
\end{tikzpicture}
\end{document}
