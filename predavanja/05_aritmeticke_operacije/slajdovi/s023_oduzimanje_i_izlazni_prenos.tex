\begin{frame}[label=05-s023]{Oduzimanje u drugom komplementu}
\fontsize{10}{12}\selectfont\setlength{\parskip}{1pt}\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
Postupak ne zavisi od odnosa $a$ i $b$.
\[\begin{gathered}a,b\geq0:\quad a-b\equiv a+(2^n-b)=2^n-(b-a)=2^n+(a-b)\end{gathered}\]
\begin{columns}[t,onlytextwidth]
\column{.49\textwidth}\centering
$a<b$\\$a+(2^n-b)=2^n-(b-a)$\\
Negativan rezultat u drugom komplementu; nema izlaznog prenosa.
\column{.48\textwidth}\centering
$a>b$\\$a+(2^n-b)=2^n+(a-b)$\\
Pozitivan rezultat; javlja se izlazni prenos.
\end{columns}\par\vspace{1mm}
