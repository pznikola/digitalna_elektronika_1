\begin{frame}[label=05-s003]{Poređenje od viših bitova}
\begin{itemize}
\item Komponentu možemo proširiti na četiri bita istom logikom.
\item Ako je $A1>B1$, onda je $A>B$; niži biti ne utiču na rezultat.
\item Ako je $A1<B1$, onda je $A<B$; niži biti ne utiču na rezultat.
\item Samo ako je $A1=B1$, posmatramo niže bite.
\end{itemize}
\medskip
\Rezultat{Bite $Ai,Bi$ poredimo samo ako su svi biti više težine jednaki.}
\medskip
$Ai>Bi\Rightarrow A>B$;\quad $Ai<Bi\Rightarrow A<B$.\par
$Ai=Bi$: nastavljamo ka bitima manje težine.
\end{frame}
