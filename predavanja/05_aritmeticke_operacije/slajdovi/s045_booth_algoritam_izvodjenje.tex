\begin{frame}[label=05-s045]{Boothov algoritam — izvođenje}
\fontsize{10}{12}\selectfont\setlength{\parskip}{0pt}\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}\setlength{\parskip}{0pt}\setlength{\jot}{0pt}\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}
Za označeno množenje često se koristi \alert{Booth’s Algorithm}.
\[D=D_{n-1}2^{n-1}+D^*,\qquad D_t=\begin{cases}D^*,&D_{n-1}=0,\\-D_{n-1}2^{n-1}+D^*,&D_{n-1}=1.\end{cases}\]

\alert{Za proizvod} $b*a$ \alert{formirajmo parcijalne proizvode na sledeći način}
\[a=(a_{31}a_{30}a_{29}a_{28}\ldots a_3a_2a_1a_0)_2\]
\begin{columns}[c,onlytextwidth]
\column{.67\textwidth}\centering
$\begin{gathered}(a_{-1}-a_0)\mathbin{\mathrm{x}}b\mathbin{\mathrm{x}}2^0\\(a_0-a_1)\mathbin{\mathrm{x}}b\mathbin{\mathrm{x}}2^1\\(a_1-a_2)\mathbin{\mathrm{x}}b\mathbin{\mathrm{x}}2^2\\\ldots\\(a_{29}-a_{30})\mathbin{\mathrm{x}}b\mathbin{\mathrm{x}}2^{30}\\(a_{30}-a_{31})\mathbin{\mathrm{x}}b\mathbin{\mathrm{x}}2^{31}\end{gathered}$
\column{.28\textwidth}$\textcolor{DEcrvena}{a_{-1}=0}$
\end{columns}
i saberimo
\[=b\mathbin{\mathrm{x}}\textcolor{DEcrvena}{(-a_{31}2^{31}+a_{30}2^{30}+\ldots+a_1 2^1+a_0 2^0)}\]
\end{frame}
