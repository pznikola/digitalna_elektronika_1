\begin{frame}[label=05-s027]{Sabiranje u prvom komplementu}
\fontsize{10}{12}\selectfont\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}\setlength{\parskip}{0pt}
\[a,b\geq0\]
\[\begin{gathered}a+(-b)\equiv a+(2^n-1-b)=2^n-1-(b-a)=\textcolor{DEisticanje}{2^n}\textcolor{DEcrvena}{-1}+(a-b)\\(-a)+b\equiv(2^n-1-a)+b=2^n-1-(a-b)=\textcolor{DEisticanje}{2^n}\textcolor{DEcrvena}{-1}+(b-a)\\[1mm]a+b\equiv a+b\\(-a)+(-b)\equiv(2^n-1-a)+(2^n-1-b)=\textcolor{DEisticanje}{2^n}\textcolor{DEcrvena}{-1}+2^n-1-(a+b)\end{gathered}\]
\begin{tabularx}{\textwidth}{@{}p{2.3cm}X@{}}
1. i 2. slučaj:&Ako su operandi različitog znaka $a_{n-1}\neq b_{n-1}$ nema prekoračenja,\\[1mm]
3. i 4. slučaj:&Ako su operandi istog znaka $a_{n-1}=b_{n-1}$\newline nema prekoračenja ako je i rezultat istog znaka\\[1mm]
1. i 2. slučaj:&Pozitivan rezultat: izlazni prenos zahteva dodavanje još 1.\\[1mm]
4. slučaj:&Izlazni prenos zahteva dodavanje još 1.
\end{tabularx}\par
\begin{center}\alert{U opštem slučaju na rezultat treba dodati keri bit.}\end{center}
