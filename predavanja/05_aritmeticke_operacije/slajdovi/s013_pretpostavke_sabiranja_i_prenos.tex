\begin{frame}[label=05-s013]{Sabiranje — pretpostavke i prenos}
\fontsize{10}{12}\selectfont\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}\setlength{\parskip}{1pt}
\begin{enumerate}
\item Fiksna tačka je usaglašena u operandima; računamo celobrojne zapise.
\item Operandi i rezultat imaju ograničen broj cifara $n$.
\item Registri sadrže potrebne vodeće i prateće nule.
\end{enumerate}
\[a_i,b_i,s_i\in\{0,1,2,\ldots,r-2,r-1\}\]
\begin{columns}[c,onlytextwidth]
\column{.40\textwidth}\centering\input{slike/tikz/s013_opste_sabiranje.tex}
\column{.57\textwidth}
$C$ — \textit{carry}: prenos 0 ili 1 u svakoj osnovi.\par
Sabiranje radimo u osnovi brojnog sistema.
\end{columns}
\[\begin{aligned}(a_i+b_i+C_{i-1})_{\max}&=(r-1)+(r-1)+1=2r-1\\&=\underbrace{r}_{C_i=1}+\underbrace{(r-1)}_{s_i=r-1}\end{aligned}\]
\[s_n=0+0+C_{n-1}=0\ \text{ili}\ 1\]
\alert{$s_n=1$: rezultat ne može da stane u $n$ cifara.}
\end{frame}
