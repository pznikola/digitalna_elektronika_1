\begin{frame}[label=06-s026]{Hemingovo rastojanje}
Hamming distance — Hemingovo rastojanje.\par\medskip
\textbf{Rastojanje dve kodne reči}: broj različitih bita.
\par\medskip
\textbf{Rastojanje koda}: najmanje rastojanje između dve različite kodne reči.
\medskip
\begin{columns}[c,onlytextwidth]
\column{.30\textwidth}\centering
\begin{tabular}{c}10\textcolor{DEcrvena}{0}1\textcolor{DEcrvena}{1}\\[1mm]10\textcolor{DEcrvena}{1}1\textcolor{DEcrvena}{0}\\[2mm]$d=2$\end{tabular}
\column{.66\textwidth}
Binarni kod: $d=1$.\par\medskip
Binarni kod sa bitom parnosti: $d=2$.
\end{columns}\bigskip
Na osnovu teorije kodovanja:
\[\begin{aligned}\text{Otkrivanje }k\text{ grešaka:}\quad&d\ge k+1,\\[2mm]\text{Korekcija }l\text{ grešaka:}\quad&d\ge2l+1.\end{aligned}\]
