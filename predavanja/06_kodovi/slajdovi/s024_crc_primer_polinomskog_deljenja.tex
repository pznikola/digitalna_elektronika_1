\begin{frame}[label=06-s024]{CRC — primer polinomskog deljenja}
\fontsize{10}{12}\selectfont\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}\setlength{\jot}{2pt}
Poruka: $11100110\equiv X^7+X^6+X^5+X^2+X$.\par
Generator polinom: $X^4+X^3+1$.
\begin{align*}
\frac{(X^7+X^6+X^5+X^2+X)X^4}{X^4+X^3+1}
&=\frac{X^{11}+X^{10}+X^9+X^6+X^5}{X^4+X^3+1}\\
&=X^7+X^5+X^4+X^2+X+\frac{\textcolor{DEcrvena}{X^2+X}}{X^4+X^3+1}
\end{align*}
Ostatak: $\textcolor{DEcrvena}{X^2+X}\equiv0110$.\par
Poruka koja se šalje: $11100110\textcolor{DEcrvena}{0110}$.
\par\smallskip
Provera množenjem, \alert{po modulu 2}:
\begin{align*}
&(X^7+X^5+X^4+X^2+X)(X^4+X^3+1)\\
&\qquad=X^{11}+X^{10}+X^9+X^6+X^5+\textcolor{DEcrvena}{(X^2+X)}.
\end{align*}
\par\smallskip
Dodavanjem ostatka: $\textcolor{DEcrvena}{(X^2+X)+(X^2+X)=0}$.
