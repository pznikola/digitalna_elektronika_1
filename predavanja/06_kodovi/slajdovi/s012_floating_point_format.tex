\begin{frame}[label=06-s012]{Floating point format}
Standard IEEE 754\par
IEEE — Institute of Electrical and Electronics Engineers
\[V=(-1)^{znak}\times\mathit{mantisa}\times\mathit{osnova}^{eksponent}\]
\smallskip
\begin{columns}[T,onlytextwidth]
\column{.29\textwidth}\centering\textbf{Znak}
\[S\in\{0,1\}\]
$S=0$ — pozitivni\\$S=1$ — negativni
\column{.32\textwidth}\centering\textbf{Mantisa}
\[\mathit{mantisa}=1.F\]
$F$ — zapisana frakcija\par\smallskip
1 — skriveni bit; ne postoji u zapisu
\column{.32\textwidth}\centering\textbf{Eksponent}
\[\mathit{eksponent}=E-B\]
$E$ se zapisuje sa $k$ bita
\[B=2^{k-1}-1\]
\end{columns}\medskip
\centering\alert{Zapis — S\quad E\quad F}
