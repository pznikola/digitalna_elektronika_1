\begin{center}FLOATING POINT FORMAT\end{center}
Standard IEEE 754\par IEEE - Institute of Electrical and Electronics Engineers
\[V=(-1)^{znak}\times\mathit{mantisa}\times\mathit{osnova}^{eksponent}\]
\[\begin{gathered}\mathit{znak}=S\in\{0,1\}\\S=0\to\mathit{pozitivni}\\S=1\to\mathit{negativni}\end{gathered}\]
\[\mathit{mantisa}=1.F\]
\begin{center}$F$ – frakcija koja je u zapisu\\1 – skriveni bit koji se podrazumeva, ne postoji u zapisu\end{center}
\[\mathit{eksponent}=E-B\]
\begin{center}$E$ se zapisuje sa $k$ bita\end{center}
\[B=2^{k-1}-1\]
