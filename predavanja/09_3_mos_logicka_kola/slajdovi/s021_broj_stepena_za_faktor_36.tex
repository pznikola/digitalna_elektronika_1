\begin{frame}[label=09_3-s021]{Broj stepena za faktor 3,6}
% element: 09_3-s021:e01
% element: 09_3-s021:e02

\[f=\sqrt[N]F\quad\Longrightarrow\quad
N=\frac{\ln F}{\ln f}=\frac{\ln(C_L/C_{G,1})}{\ln(3.6)}\]
Za $N$ se bira najbliži neparan broj.
\[f_1=1,\qquad f_2=\lfloor3.6+0.5\rfloor=4,\qquad
f_3=\lfloor3.6^2+0.5\rfloor=13,\ldots\]
$\lfloor x\rfloor$ — ceo deo; $\lfloor x+0.5\rfloor$ — najbliži ceo broj.
\[t_p=Nt_{p0}+t_{p0}\sum_{i=1}^N\frac{f_{i+1}}{f_i}\]
\end{frame}
