\begin{frame}[label=09_3-s023]{Nebaferisana i dvostruko baferisana kola}
% element: 09_3-s023:e01
% element: 09_3-s023:e02
% element: 09_3-s023:e03
\begin{itemize}
\item Lanac bez velikog spoljnog opterećenja ima znatno veće interno kašnjenje od jediničnog invertora.
\item Ako se opterećenje smanji, prvobitno dimenzionisanje više nije optimalno.
\item \alert{Nebaferisana kola:} malo izlazno opterećenje i malo kašnjenje.
\item \alert{Dvostruko baferisana kola:} dva izlazna invertora veličine $4\times$ i $16\times$, za opterećenje $64\times$.
\end{itemize}
\end{frame}
