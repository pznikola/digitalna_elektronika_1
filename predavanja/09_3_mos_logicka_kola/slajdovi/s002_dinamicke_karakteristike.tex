\begin{frame}[label=09_3-s002]{Dinamičke karakteristike CMOS invertora}
% element: 09_3-s002:e01
% element: 09_3-s002:e02
\begin{itemize}
\item MOS tranzistori imaju velike interne kapacitivnosti.
\item Kapacitivnosti opterećuju sam invertor i prethodni stepen.
\item Parazitne kapacitivnosti vodova smatramo relativno konstantnim.
\item Dimenzije tranzistora utiču i na \alert{statičke} i na \alert{dinamičke} karakteristike.
\end{itemize}
\end{frame}
