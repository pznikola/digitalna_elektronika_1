\begin{frame}[label=09_3-s041]{Optimizacija dva susedna kola}
% element: 09_3-s041:e01
% element: 09_3-s041:e02
% element: 09_3-s041:e03
Deo kašnjenja kroz prva dva kola:
\[\begin{aligned}
t_{p,i(i+1)}={}&\tau_{NAND}\left(\frac{C_{g,i+1}}{C_{g,i}}+\gamma_{NAND}\right)\\
&+\tau_{inv}\left(\frac{C_{g,i+2}}{C_{g,i+1}}+\gamma_{inv}\right).
\end{aligned}\]
Minimum izborom $C_{g,i+1}$:
\[\frac{\partial t_{p,i(i+1)}}{\partial C_{g,i+1}}
=\frac{\tau_{NAND}}{C_{g,i}}-\frac{\tau_{inv}C_{g,i+2}}{(C_{g,i+1})^2}=0\]
\[\alert{\tau_{NAND}FO_i=\tau_{inv}FO_{i+1}},\qquad
FO_i=\frac{C_{g,i+1}}{C_{g,i}},\quad FO_{i+1}=\frac{C_{g,i+2}}{C_{g,i+1}}\]
CMOS fanout ograničavaju kapacitivnosti i kašnjenje.
\end{frame}
