\begin{frame}[label=10-s088]{ECL: precizan izbor elemenata}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
Za $V_R=V_{CC}-R_CI_E/2$ i T2 koji provodi celu struju:
\[\begin{aligned}V_{B2}&=V_R=V_{CC}-R_CI_E/2\\V_{E2}&=V_{B2}-V_{BE2}=V_{CC}-R_CI_E/2-V_{BE}\\V_{C2}&=V_{CC}-R_CI_E\end{aligned}\]
\bigskip\Oznaka{T2 mora ostati aktivan}
\[V_{CE2}=V_{BE}-R_C\frac{I_E}2>V_{CES}\]
\Rezultat{Precizan izbor otpornika i struje izvora: npr. $779\,\Omega$.}
\end{frame}
