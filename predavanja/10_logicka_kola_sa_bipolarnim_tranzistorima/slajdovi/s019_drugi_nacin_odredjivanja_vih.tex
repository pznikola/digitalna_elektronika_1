\begin{frame}[label=10-s019]{Drugi način određivanja visokog praga}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
U zasićenju:
\[I_{B1}=\frac{V_I-V_{BES}}{R_B},\qquad I_{C1}=\frac{V_{CC}-V_{CES}}{R_C}\]
Smanjujemo $V_I$ do $\beta_FI_{B1}=I_{C1}$:
\[\beta_F\frac{V_{I2}-V_{BES}}{R_B}=\frac{V_{CC}-V_{CES}}{R_C}\]
\[V_{I2}=\frac{R_B}{\beta_FR_C}(V_{CC}-V_{CES})+V_{BES}\]
U poređenju sa prethodnim rezultatom:
\[V_{I1}=\frac{R_B}{\beta_FR_C}(V_{CC}-V_{CES})+V_{BE}\]
\Rezultat{\centering $V_{I2}-V_{I1}=V_{BES}-V_{BE}$.}
\end{frame}
