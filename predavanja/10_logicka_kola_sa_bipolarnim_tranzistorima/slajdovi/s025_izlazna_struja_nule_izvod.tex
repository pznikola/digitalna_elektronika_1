\begin{frame}[label=10-s025]{Izlazni strujni kapacitet nule}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
Nizak izlaz: tranzistor ostaje u zasićenju.
\[I_{C1}=I_{RC}+I_O\]
\[\beta_FI_{B1}>I_{C1}=I_{RC}+I_O\]
\[I_O<\beta_FI_{B1}-I_{RC}\]
Izlazna kontura:
\[I_{RC}=\frac{V_{CC}-V_O}{R_C}=\frac{V_{CC}-V_{CES}}{R_C}\]
\Rezultat{\centering $\displaystyle I_O<\beta_FI_{B1}-I_{RC}=\beta_FI_{B1}-\frac{V_{CC}-V_{CES}}{R_C}$.}
\end{frame}
