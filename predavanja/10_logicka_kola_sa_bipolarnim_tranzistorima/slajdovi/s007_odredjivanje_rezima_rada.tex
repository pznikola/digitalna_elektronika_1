\begin{frame}[label=10-s007]{Određivanje režima rada}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
\begin{enumerate}
\item Pretpostavi režim rada tranzistora.
\item Zameni tranzistor modelom za taj režim.
\item Izračunaj napone i struje u kolu.
\item Proveri uslove, npr. $\beta_F I_B>I_C$ za zasićenje.
\item Ako pretpostavka nije ispravna, vrati se na korak 1.
\end{enumerate}\bigskip
\Rezultat{Pretpostavka $\longrightarrow$ model $\longrightarrow$ proračun $\longrightarrow$ provera.}
\end{frame}
