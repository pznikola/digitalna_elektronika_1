\begin{frame}[label=10-s013]{Ulazak u aktivni režim}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
Povećanje $V_I$ daje malu baznu i kolektorsku struju.
\[V_{CE1}=V_{CC}-R_CI_{RC}=V_{CC}-R_CI_C\]
\bigskip
\Oznaka{Pretpostavka: aktivan režim}\medskip
Za male bazne struje tranzistor radi u aktivnom režimu.\par\bigskip
\Oznaka{Provera pretpostavke zasićenja}\medskip
\[V_{CE1}=V_{CC}-R_CI_{RC}>V_{CES}\]
Pretpostavka zasićenja nije ispravna.
\end{frame}
