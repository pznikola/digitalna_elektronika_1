\[v_O(t_0^+)=v_C(t_0^+)=v_C(t_0^-)=V_L\]
\[v_{RB}(t)+V_{BES}-V_{CES}-v_{RC}(t)=0\]

\[i_{RB}(\infty)+i_{RC}(\infty)=0\]
\[i_{RC}(\infty)=\frac{V_{BES}-V_{CES}}{R_B+R_C}\]
\[\begin{aligned}v_O(\infty)&=V_{CC}-R_C i_{RC}(\infty)-V_{CES}-V_D\\&=V_{CC}-V_{CES}-V_D-R_C\frac{V_{BES}-V_{CES}}{R_B+R_C}\end{aligned}\]

\[v_o(t)=v_o(\infty)+(v_o(t_0^+)-v_o(\infty))e^{-\frac{t-t_0}{\tau}}\quad za\ t\geq t_0\]
