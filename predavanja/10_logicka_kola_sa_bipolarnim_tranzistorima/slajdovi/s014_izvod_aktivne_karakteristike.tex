\begin{frame}[label=10-s014]{Aktivna karakteristika RTL kola}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
Za neopterećeno kolo i aktivan režim:
\[V_I-R_BI_{RB}-V_{BE1}=0,\qquad V_O=V_{CC}-R_CI_{RC}\]
\[I_B=I_{RB}=\frac{V_I-V_{BE1}}{R_B}\]
\[V_O=V_{CC}-R_CI_{RC}=V_{CC}-R_CI_R=V_{CC}-R_C\beta_FI_B\]

\Rezultat{\centering $\displaystyle V_O=V_{CC}-\beta_F\frac{R_C}{R_B}(V_I-V_{BE})$}\medskip
U aktivnom modelu $V_{BE}=0.6\,\mathrm{V}$:
\[V_O=V_{CC}-\beta_F\frac{R_C}{R_B}(V_I-0.6\,\mathrm{V})\]
\end{frame}
