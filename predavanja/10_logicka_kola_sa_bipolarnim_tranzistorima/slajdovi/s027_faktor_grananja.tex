\begin{frame}[label=10-s027]{Faktor grananja RTL kola}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
$V_I=V_{OH\min}>V_{IH}$? U katalog za minimalnu ulaznu jedinicu: $V_{OH\min}$.
\[I_O<\beta_F\frac{V_{OH\min}-V_{BES}}{R_B}-\frac{V_{CC}-V_{CES}}{R_C}=I_{OL}\]
Za $I_{OL}$: manje $R_B$, veće $R_C$.\par\bigskip
\[N_L=\frac{I_{OL}}{I_{IL}}=\infty\]
\[N_H=\frac{|I_{OH}|}{I_{IH}}=\frac{R_B}{R_C}\frac{V_{CC}-V_{OH\min}}{V_{CC}-V_{BES}}\sim\frac{R_B}{R_C}\]
\Rezultat{Za veći $N_H$: veće $R_B$, manje $R_C$.\quad $N=\min(N_L,N_H)$.}
\end{frame}
