\begin{frame}[label=10-s026]{Nulti strujni kapacitet na granici}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}\setlength{\abovedisplayshortskip}{3pt}\setlength{\belowdisplayshortskip}{3pt}
Najmanja bazna struja za $V_I=V_{IH}$:
\[I_{B\min}=\frac{V_{IH}-V_{BES}}{R_B}\]
\[I_O<\beta_FI_{B\min}-\frac{V_{CC}-V_{CES}}{R_C}=I_{OL}\]
Za granicu određenu iz zasićenja:
\[V_{I2}=V_{IH}=\frac{R_B}{\beta_FR_C}(V_{CC}-V_{CES})+V_{BES}\]
\[\begin{aligned}I_O&<\beta_F\frac{\frac{R_B}{\beta_FR_C}(V_{CC}-V_{CES})+V_{BES}-V_{BES}}{R_B}\\
&\phantom{{}<}\ -\frac{V_{CC}-V_{CES}}{R_C}\textcolor{DEcrvena}{=0=I_{OL}}.
\end{aligned}\]
\end{frame}
