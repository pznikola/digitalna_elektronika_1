\begin{frame}[label=09_2-s045]{Odnos širina u linearnom modelu}
\textbf{Za zadato $V_S$}\[\frac{W_p}{W_n}=\frac{\mu_nV_{DSnsat}\left(V_S-V_{Tn}-\frac{V_{DSnsat}}2\right)}{\mu_pV_{DSpsat}\left(V_S-V_{DD}-V_{Tp}-\frac{V_{DSpsat}}2\right)}\]
Za $V_S=V_{DD}/2$ i $v_{satn}=-v_{satp}$:\[\frac{W_p}{W_n}=\frac{\frac{V_{DD}}2-V_{Tn}-\frac{V_{DSnsat}}2}{\frac{V_{DD}}2+V_{Tp}+\frac{V_{DSpsat}}2}\]
Za dugačak kanal, $V_{DSnsat}>V_{DD}/2-V_{Tn}$:\[\frac{W_p}{W_n}=\frac{V_{DSnsat}}{|V_{DSpsat}|}=\frac{\mu_n}{\mu_p}\]

\end{frame}
