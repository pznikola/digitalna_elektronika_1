\begin{frame}[label=09_2-s046]{Normalizovane širine tranzistora}
\[\frac{W_p}{W_n}=\frac{\mu_n}{\mu_p}\]
\begin{itemize}
\item N minimalne geometrije: normalizovano $W_n=1$.
\item Normalizovano $W_p$: ceo broj, eventualno X.5.
\end{itemize}
\[\frac{W_p}{W_n}=\frac{\mu_n}{\mu_p}\approx\frac{\textit{ceo broj P}}{\textit{ceo broj N}}\]
Za poznato $W_n=(\text{ceo broj N})\times L$, npr. tehnologija 10 nm:\[W_p=\textit{ceo broj}\left(\frac{\mu_n}{\mu_p}\times(\textit{ceo broj N})\right)\times L\]

\end{frame}
