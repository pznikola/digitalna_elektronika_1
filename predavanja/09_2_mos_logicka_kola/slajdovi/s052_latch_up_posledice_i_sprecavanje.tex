\begin{frame}[label=09_2-s052]{Latch-up: posledice i sprečavanje}
\begin{itemize}
\item Velika struja može uništiti komponentu.
\item Ograničena struja izvora može oboriti napajanje i zaustaviti ispravan rad.
\end{itemize}
\vspace{3mm}\alert{Ako se lečap pojavio, ukloniti napon napajanja.}
\par\vspace{3mm}
\textbf{Smanjivanje rizika}
\begin{itemize}
\item Sporije izlazne ivice: duže vreme uspona i pada.
\item Prstenovi oko P i N oblasti, povezani na napajanje odnosno masu.
\end{itemize}
\end{frame}
