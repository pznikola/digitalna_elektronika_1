\begin{frame}[label=09_2-s038]{Podešavanje tačke prebacivanja CMOS invertora}
\[\begin{aligned}&k_p\frac1{1+\frac{V_I-V_{DD}-V_{Tp}}{L_pE_{Cp}}}(V_I-V_{DD}-V_{Tp})^2\left(1+\lambda_p(V_0-V_{DD})\right)\\&\quad=k_n\frac1{1+\frac{V_I-V_{Tn}}{L_nE_{Cn}}}(V_I-V_{Tn})^2(1+\lambda_nV_O)\end{aligned}\]
\begin{enumerate}
\item Pojačanje u trećoj oblasti je veliko, konačno.
\item Prag odlučivanja nalazi se u toj oblasti.
\item Odnos $k_n/k_p$ podešava vrednost praga.
\end{enumerate}
Za $\lambda_n=\lambda_p=0$ posmatramo tri slučaja:
\[V_S\ll E_CL\qquad V_S\sim E_CL\qquad V_S\gg E_CL\]
\end{frame}
