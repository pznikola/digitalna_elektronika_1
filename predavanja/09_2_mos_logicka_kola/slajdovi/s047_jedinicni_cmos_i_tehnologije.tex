\begin{frame}[label=09_2-s047]{Jedinični CMOS invertor i tehnološke dimenzije}
\begin{itemize}
\item Jedinični invertor: reprezentativan primer tehnologije.
\item Oznaka tehnologije predstavlja minimalnu dimenziju.
\item Primer 45 nm: minimalni $L$ i $W$ su 45 nm.
\item Dužine i širine: celobrojni umnošci minimuma.
\end{itemize}
