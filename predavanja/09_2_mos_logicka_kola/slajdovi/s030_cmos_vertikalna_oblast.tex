\begin{frame}[label=09_2-s030]{CMOS: veliko pojačanje u trećoj oblasti}
\begin{columns}[T,onlytextwidth]\column{.24\textwidth}\centering \input{slike/tikz/s030_cmos_invertor.tex}\column{.72\textwidth}\textbf{Oba tranzistora u zasićenju; dugačak kanal:}\[\text{N tranzistor u zasićenju:}\quad V_O\ge V_I-V_{Tn}\]
\[\text{P tranzistor u zasićenju:}\quad V_O\le V_I-V_{Tp}\]
\textbf{Ravnoteža za model kratkog kanala:}\[\begin{aligned}&\frac{k_p(V_I-V_{DD}-V_{Tp})^2}{1+(V_I-V_{DD}-V_{Tp})/(L_pE_{Cp})}\\=\frac{k_n(V_I-V_{Tn})^2}{1+(V_I-V_{Tn})/(L_nE_{Cn})}\end{aligned}\]

Bez modulacije: beskonačno pojačanje; $V_S$ je u ovoj oblasti.\end{columns}
