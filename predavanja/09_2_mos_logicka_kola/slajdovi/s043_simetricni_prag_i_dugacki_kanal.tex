\begin{frame}[label=09_2-s043]{Kratak kanal: prag na polovini napajanja}
$V_{Tn}=-V_{Tp}$ i $V_S=V_{DD}/2$:\[\color{DEcrvena}\frac{W_p}{W_n}=\frac{\frac{V_{DD}}2-V_{Tn}+|L_pE_{Cp}|}{\frac{V_{DD}}2-V_{Tn}+L_nE_{Cn}}\]
\textbf{Ako zasićenje brzine još nije nastupilo}\[\left(\frac{V_{DD}}2-V_{Tn}\right)\ll L_nE_{Cn}\quad\textit{i}\quad\left|\frac{V_{DD}}2+V_{Tp}\right|\ll|L_pE_{Cp}|\]
Granični oblik za dugačak kanal:\[\color{DEcrvena}\frac{W_p}{W_n}=\frac{|L_pE_{Cp}|}{L_nE_{Cn}}=\frac{L_p\frac{2v_{satp}}{\mu_p}}{L_n\frac{2v_{satn}}{\mu_n}}=\frac{\mu_n}{\mu_p}\]

\end{frame}
