\begin{frame}[label=09_2-s057]{Napajanje, kašnjenje i disipacija}
\[t_p=0.69R_{eq}C_{eq}\]
Manje kašnjenje zahteva manji $R_{eq}$ ili $C_{eq}$.\[R_p\approx\frac34\frac{V_{DD}}{I_{Dpsat}}\left(1-\frac79|\lambda_p|V_{DD}\right)\]
\[I_{Dpsat}=\frac{k_p}2\frac1{1+\frac{-V_{DD}-V_{Tp}}{E_{Cp}L_p}}(-V_{DD}-V_{Tp})^2\]
\textbf{Uticaj napajanja}
$R_{eq}\sim1/V_{DD}$.
Veće napajanje smanjuje kašnjenje; disipacija raste sa trećim stepenom.
\par\medskip\textbf{Voltage frequency scaling:} smanjenje učestanosti i napajanja kada puna brzina nije potrebna.
\end{frame}
