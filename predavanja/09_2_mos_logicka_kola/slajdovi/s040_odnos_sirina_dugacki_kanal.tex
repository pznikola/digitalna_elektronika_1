\begin{frame}[label=09_2-s040]{Dimenzionisanje za željenu tačku prebacivanja}
\begin{columns}[T,onlytextwidth]\column{.46\textwidth}\textbf{Željeni prag}\[r=\frac{V_S-V_{Tn}}{V_{DD}-V_S+V_{Tp}}\]
Za $V_S=V_{DD}/2$:\[r=\frac{\frac{V_{DD}}2-V_{Tn}}{\frac{V_{DD}}2+V_{Tp}}\]
$V_{Tn}=-V_{Tp}$ daje $r=1$.\[\sqrt{\frac{k_p}{k_n}}=1\quad\Rightarrow\quad\frac{k_p}{k_n}=1\]
\column{.50\textwidth}\textbf{Podešavanje širinama}\par\medskip Odnos određujemo širinama kanala.\par\medskip $C_{oxp}=C_{oxp}$, $L_p=L_n$.\[\color{DEcrvena}\frac{W_p}{W_n}=\frac{\mu_n}{\mu_p}\]
U opštem slučaju:\[\color{DEcrvena}\frac{W_p}{W_n}=\frac{\mu_n}{\mu_p}\left(\frac{V_S-V_{Tn}}{V_{DD}-V_S+V_{Tp}}\right)^2\]
\end{columns}
\end{frame}
