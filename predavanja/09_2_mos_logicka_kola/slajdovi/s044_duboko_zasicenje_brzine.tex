\begin{frame}[label=09_2-s044]{Tačka prebacivanja: izraženo zasićenje brzine}
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{2pt}
$V_S\gg E_CL$: primenjujemo linearni model struje.\[\begin{aligned}&V_S-V_{Tn}>V_{DSnsat}=L_nE_{Cn}=L_n\frac{v_{satn}}{\mu_n}\\& V_S-V_{DD}-V_{Tp}<V_{DSpsat}=L_pE_{Cp}=L_p\frac{v_{satp}}{\mu_p}\end{aligned}\]
\[\begin{aligned}&W_pv_{satp}\left(V_S-V_{DD}-V_{Tp}-\frac{V_{DSpsat}}2\right)\\&=W_nv_{satn}\left(V_S-V_{Tn}-\frac{V_{DSnsat}}2\right)\end{aligned}\]
\[\frac{-W_pv_{satp}}{W_nv_{satn}}=\frac{W_p|v_{satp}|}{W_nv_{satn}}=r\]
\[V_S=\frac{V_{Tn}+\frac{V_{DSnsat}}2+r\left(V_{DD}+V_{Tp}+\frac{V_{DSpsat}}2\right)}{1+r}\]

\end{frame}
