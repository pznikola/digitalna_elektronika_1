\begin{frame}[label=01-s030]{Aktivan nivo: I i ILI}
% element: 01-s030:e01
U pozitivnoj logici \textbf{aktivan nivo na izlazu} I i ILI kola je logička \(1\).
% element: 01-s030:e02
\ETFPoredjenje{I (AND)}{Izlaz je \(1\) kada su \textbf{svi} ulazi \(1\).\par\[y=x_1x_2\cdots x_n\]}{ILI (OR)}{Izlaz je \(1\) kada je \textbf{bar jedan} ulaz \(1\).\par\[y=x_1+x_2+\cdots+x_n\]}
% element: 01-s030:e03
\vspace{5mm}\Rezultat{Kada je logički uslov ispunjen, aktivan nivo na izlazu je \(1\).}

\end{frame}
