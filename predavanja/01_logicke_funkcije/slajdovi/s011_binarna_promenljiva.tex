\begin{frame}[label=01-s011]
Umesto tačno netačno koristićemo
\begin{itemize}
\item logičku nulu „0“
\item logičku jedinicu „1“
\end{itemize}

Pridruživanje

\hspace*{5mm} Pridružujemo promenljivu \(P\) stanju prekidača
\begin{itemize}
\item prekidač zatvoren – \(P=\text{„1“}\)
\item prekidač nije zatvoren odnosno otvoren je – \(P=\text{„0“}\)
\end{itemize}
Promenljiva koja je pridružena prekidaču \(P\) može imati samo dva stanja\\
ili stanje logičke jedinice ili stanje logičke nule.
\[
A=\{0,1\}\qquad P\in A
\]
\centering \(P\) – binarna promenjiva
\end{frame}
