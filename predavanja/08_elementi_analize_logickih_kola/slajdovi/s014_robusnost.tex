\begin{frame}[label=08-s014]{Robusnost}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}\textbf{Logička nula}\par\medskip
Smetnja sme da povisi $V_{OL}$ do $V_{IL}$.
\[\alert{NM_L=V_{IL}-V_{OL}}\]
\column{.47\textwidth}\textbf{Logička jedinica}\par\medskip
Smetnja sme da snizi $V_{OH}$ do $V_{IH}$.
\[\alert{NM_H=V_{OH}-V_{IH}}\]
\end{columns}\medskip
NM — noise margin, margina šuma.\par\medskip
\[NM=\min(NM_L,NM_H)\]
Primer: $NM_H=2\,\mathrm V$, $NM_L=1\,\mathrm V$ $\Rightarrow NM=1\,\mathrm V$.\par\medskip
Provera: $NM_L+NM_H\leq V_{CC}$.
\end{frame}
