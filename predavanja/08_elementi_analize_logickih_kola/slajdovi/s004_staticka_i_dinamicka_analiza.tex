\centering
Analiza logičkih kola se odnosi na analizu\par
\alert{Statičkih karakteristika\\Dinamičkih karakteristika}\par\bigskip
\raggedright
Statičke karakteristike
\begin{quote}\upshape Kako se logičko kolo ponaša za različite nivoe ulaznih signala.\\
Ne postoji vremenska dimenzija prilikom analize.\end{quote}
\hfill $b=f(a)$\hspace{1cm}\par\bigskip
Dinamičke karakteristike
\begin{quote}\upshape Kako se kolo ponaša u vremenskom domenu kada se ulazni signali menjaju.\\
Kakva je vremenska zavisnost izlaznih signala u odnosu na promenu ulaznih signala.\end{quote}
\hfill $b(t)=f(a(t))$\hspace{1cm}
\end{frame}
