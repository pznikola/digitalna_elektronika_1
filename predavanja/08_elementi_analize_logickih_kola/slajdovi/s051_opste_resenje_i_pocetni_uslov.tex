\begin{frame}[label=08-s051]{Drugačije viđenje - Opšte rešenje}
\setlength{\parskip}{2pt}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
\[u_o(t)=K_1e^{-\frac{t-t_0}{\tau}}+K_2\]
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}\textbf{Oznake}\par\medskip
$u_x(\infty)=\displaystyle\lim_{t\to\infty}u_x(t)$\par\medskip
$u_x(t_0^-)$ — neposredno pre promene.\par\medskip
$u_x(t_0^+)$ — neposredno posle promene.
\column{.53\textwidth}\textbf{1. Početni uslov}
\begin{align*}
u_o(t_0^+)&=K_1e^{-\frac{t_0^+-t_0}{\tau}}+K_2\\
&=K_1+K_2
\end{align*}
\[K_1+K_2=u_o(t_0^+)\]
\end{columns}
\end{frame}
