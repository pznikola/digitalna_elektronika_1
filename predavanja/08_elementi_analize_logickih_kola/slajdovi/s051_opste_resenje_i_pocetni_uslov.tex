\begin{frame}[label=08-s051]{Dugačije viđenje - Opšte rešenje}
\gusttekst
\[u_o(t)=K_1e^{-\frac{t-t_0}{\tau}}+K_2\]
Oznake\par
$\displaystyle\lim_{t\to\infty}u_x(t)=u_x(\infty)$ - napon u beskonačnosti,\par\medskip
$u_x(t_0^-)\quad za\quad(t_0-t_0^-)\to0$ - napon u trenutku neposredno pre trenutka $t_0$, pre dejstva promene\par\medskip
$u_x(t_0^+)\quad za\quad(t_0^+-t_0)\to0$ - napon u trenutku neposredno posle trenutka $t_0$, posle dejstva promene.\par\bigskip
1.
\[u_o(t_0^+)=K_1e^{-\frac{t_0^+-t_0}{\tau}}+K_2=K_1+K_2\]
onda je
\[K_1+K_2=u_o(t_0^+)\]
\end{frame}
