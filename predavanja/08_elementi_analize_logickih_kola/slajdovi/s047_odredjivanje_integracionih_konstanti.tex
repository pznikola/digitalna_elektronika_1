\begin{frame}[label=08-s047]
\setlength{\parskip}{2pt}
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
Konstante $K_1$ i $K_2$ određujemo iz početnih uslova.\par\smallskip
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}\textbf{Napon u trenutku $t_0$}
\begin{align*}
u_o(t_0)&=K_1e^{-\frac{t_0-t_0}{\tau}}+K_2\\
&=K_1+K_2\\
K_1+K_2&=U_0
\end{align*}
\column{.53\textwidth}\textbf{Nagib iz diferencijalne jednačine}
\begin{align*}
\frac{du_o(t)}{dt}&=\frac{u_i(t)-u_o(t)}{RC}\\
\frac{du_o(t_0)}{dt}&=\frac{u_i(t_0)-u_o(t_0)}{RC}\\
&=\frac{U-U_o}{\tau}
\end{align*}
\end{columns}\smallskip
Iz opšteg rešenja:
\[\frac{du_o(t)}{dt}=\frac{d\left(K_1e^{-\frac{t-t_0}{\tau}}+K_2\right)}{dt}=-\frac{K_1}{\tau}e^{-\frac{t-t_0}{\tau}}\]
\[\left.\frac{du_o(t)}{dt}\right|_{t=t_0}=-\frac{K_1}{\tau}\]
\end{frame}
