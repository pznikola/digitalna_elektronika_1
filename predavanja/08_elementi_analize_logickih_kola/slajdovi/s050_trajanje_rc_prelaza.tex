\begin{frame}[label=08-s050]
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}\textbf{Vreme uspona: 10\% do 90\%}
\begin{align*}
u_o(t_1)&=10\%\,(U-U_0)+U_0\\
u_o(t_2)&=90\%\,(U-U_0)+U_0\\
t_1&=t_0-\tau\ln(0.9)\\
t_2&=t_0-\tau\ln(0.1)\\
t_r&=\tau\ln(9)\approx\alert{2.2\tau}
\end{align*}
\column{.47\textwidth}\textbf{Kašnjenje: nivo 50\%}
\begin{align*}
u_o(t_1)&=50\%\,(U-U_0)+U_0\\
t_1&=t_0-\tau\ln(0.5)\\
t_p&=t_1-t_0\\
&=\tau\ln(2)\approx\alert{0.69\tau}
\end{align*}
\end{columns}\medskip
Simetrični rezultati važe za silaznu ivicu.
\end{frame}
