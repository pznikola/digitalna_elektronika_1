Pre trenutka $t_0$ ulazni napon $U_0$ traje dovoljno dugo da se uspostavi stacionarno stanje.\par\medskip
\begin{align*}
\lim_{u_C(t)\to U_0}i_R(t)&=\lim_{u_C(t)\to U_0}i_C(t)\\
&=\lim_{u_C(t)\to U_0}\frac{u_C(t)-u_o(t)}R\\
&=\lim_{u_C(t)\to U_0}\frac{u_C(t)-U_0}R=0
\end{align*}
\[i_C(t)=0\Rightarrow\frac{du_C(t)}{dt}=0\Rightarrow u_C(t)=\mathrm{const}\]
\medskip\par\centering Stacionarno stanje: $i_C(t)=0$, $u_C(t)=\mathrm{const}$.
\end{frame}
