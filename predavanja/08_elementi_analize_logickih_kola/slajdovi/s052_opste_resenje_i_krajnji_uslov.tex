\begin{frame}[label=08-s052]
2.
\[\lim_{t\to\infty}u_o(t)=u_o(\infty)\]
\[\lim_{t\to\infty}\left(K_1e^{-\frac{t-t_0}{\tau}}+K_2\right)=K_2\]
onda je\qquad $K_2=u_o(\infty)$\par
pa je i\qquad $K_1=u_o(t_0^+)-u_o(\infty)$\par\bigskip
Rešenje diferencijalne jednačine je
{\color{DEcrvena}\[u_o(t)=u_o(\infty)+\bigl(u_o(t_0^+)-u_o(\infty)\bigr)e^{-\frac{t-t_0}{\tau}}\qquad za\ t\geq t_0\]}
\end{frame}
