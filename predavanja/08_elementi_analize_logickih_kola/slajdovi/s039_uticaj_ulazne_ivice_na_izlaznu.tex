\begin{frame}[label=08-s039]
Trajanje izlazne ivice takođe zavisi od trajanja ulazne ivice.\par\medskip
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}\textbf{Silazna izlazna ivica}
\[t_{fout}=\sqrt{t_{fout0}^{2}+t_r^2}\]
\column{.47\textwidth}\textbf{Uzlazna izlazna ivica}
\[t_{rout}=\sqrt{t_{rout0}^{2}+t_f^2}\]
\end{columns}\medskip
\begin{itemize}
\item $t_{fout}$ — izmereno vreme pada na izlazu.
\item $t_r$ — izmereno vreme uspona na ulazu.
\item $t_{fout0}$ — doprinos samog kola, koji računamo.
\end{itemize}
\end{frame}
