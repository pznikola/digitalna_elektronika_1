\begin{frame}[label=04-s003]{Celobrojna i razlomljena vrednost}
% element: 04-s003:e01 — Rastavljanje sume V na CV i RV
% element: 04-s003:e02 — Zapisi celobrojnog i razlomljenog dela; značenje prefiksa 0.
% element: 04-s003:e03 — Vodeće nule celobrojnog dela sa parom ekvivalentnih zapisa
% element: 04-s003:e04 — Prateće nule razlomljenog dela sa parom ekvivalentnih zapisa
\[c_{n-1}c_{n-2}\ldots c_1c_0.\,c_{-1}c_{-2}\ldots c_{-k}\]
\[V=\sum_{i=-k}^{n-1}c_i r^i=\sum_{i=0}^{n-1}c_i r^i+\sum_{i=-k}^{-1}c_i r^i=\alert{CV+RV}\]
\begin{columns}[T,onlytextwidth]
\column{.475\textwidth}\NaslovBloka{Celobrojna vrednost — CV}
\[c_{n-1}c_{n-2}\ldots c_1c_0\]
Dodatak \alert{vodećih nula} ne menja vrednost.
\[c_{n-1}\ldots c_0\leftrightarrow00\ldots00c_{n-1}\ldots c_0\]
\column{.475\textwidth}\NaslovBloka{Razlomljena vrednost — RV}
\[0.c_{-1}c_{-2}\ldots c_{-k}\]
Celobrojna vrednost je nula.\par
Dodatak \alert{pratećih nula} ne menja vrednost.
\[0.c_{-1}\ldots c_{-k}\leftrightarrow0.c_{-1}\ldots c_{-k}00\ldots00\]
\end{columns}
\end{frame}
