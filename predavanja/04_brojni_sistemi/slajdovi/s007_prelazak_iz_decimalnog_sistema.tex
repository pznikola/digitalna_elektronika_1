\begin{frame}[label=04-s007]
\gusttekst
\alert{Prelazak iz decimalnog brojnog sistema u brojni sistem sa osnovom $r$}
\[V_{10}=\sum_{i=-k}^{n-1}c_i r^i\qquad c_i\in\{0,1,2,\ldots r-2,r-1\}\]
\[V_{10}=\sum_{i=0}^{n-1}c_i r^i+\sum_{i=-k}^{-1}c_i r^i=CV_{10}+RV_{10}\]
Celobrojna vrednost
\[CV_{10}=c_{n-1}\cdot r^{n-1}+c_{n-2}\cdot r^{n-2}+\ldots+c_1\cdot r^1+c_0\cdot r^0\]
Razlomljena vrednost
\[RV_{10}=c_{-1}\cdot r^{-1}+c_{-2}\cdot r^{-2}+\ldots+c_{-k+1}\cdot r^{-k+1}+c_{-k}\cdot r^{-k}\]
\end{frame}
