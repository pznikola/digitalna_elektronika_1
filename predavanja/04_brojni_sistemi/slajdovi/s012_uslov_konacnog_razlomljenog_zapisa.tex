\begin{frame}[label=04-s012]{Uslov konačnog razlomljenog zapisa}
% element: 04-s012:e01 — Razvijanje RVr preko težina i zajedničkog imenioca; RVr=CVr/r^k
% element: 04-s012:e02 — Jednakost razlomaka sa l decimalnih i k cifara osnove r
% element: 04-s012:e03 — Izraz za traženu celobrojnu vrednost (CVr)k
% element: 04-s012:e04 — Uslov nije uvek dostižan; nulti ostatak ili dovoljan broj cifara
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
\[\begin{aligned}RV_r&=c_{-1}r^{-1}+c_{-2}r^{-2}+\ldots+c_{-k}r^{-k}\\
&=\frac{c_{-1}r^{k-1}+c_{-2}r^{k-2}+\ldots+c_{-k}r^0}{r^k}=\frac{CV_r}{r^k}\end{aligned}\]
Decimalni razlomljeni deo ima $l$ cifara. Za konačan zapis sa $k$ cifara u osnovi $r$ mora važiti:
\[\frac{(CV_{10})_l}{10^l}=\frac{(CV_r)_k}{r^k}\quad\Longrightarrow\quad(CV_r)_k=\frac{r^k}{10^l}(CV_{10})_l.\]
\alert{Ovaj uslov nije uvek moguće zadovoljiti.}
\[(RV_{10})_{-k-1}=0\ \Rightarrow\ c_{-k-1}=0\]
Ako uslov nije dostižan, računamo \alert{dovoljan broj cifara}.
\end{frame}
